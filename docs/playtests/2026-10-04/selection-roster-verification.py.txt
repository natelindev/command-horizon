import concurrent.futures,json,subprocess,time
from pathlib import Path
base=Path('docs/playtests/2026-10-04')
godot='./tmp/tooling/godot-4.6/Godot.app/Contents/MacOS/Godot'
checks=[]
def run(name,args,limit=240,marker=False):
 log=base/('selection-roster-final-'+name+'.log')
 start=time.monotonic()
 with log.open('w') as out:
  result=subprocess.run(args,stdout=out,stderr=subprocess.STDOUT,timeout=limit)
 content=log.read_text()
 row={'suite':name,'exit':result.returncode,'script_error':'SCRIPT ERROR' in content,'load_error':'Failed to load' in content,'pass_marker':('0 failures' in content or 'PASS:' in content) if marker else None,'wall_seconds':round(time.monotonic()-start,1),'log':str(log)}
 print(json.dumps(row),flush=True)
 return row
checks.append(run('import',[godot,'--headless','--path','.','--editor','--quit']))
suites=['SelectionRosterRegression','RosterInfoRegression','OrdersRegression','SquadRegression','ProgramRegression','WeaponModeRegression','SaveRestoreRegression','SubtitleLayoutRegression','CampaignControlsRegression']
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
 futures=[pool.submit(run,name,[godot,'--headless','--path','.','--fixed-fps','60','--script','tests/headless/'+name+'.gd'],240,True) for name in suites]
 for f in concurrent.futures.as_completed(futures):
  checks.append(f.result())
  (base/'selection-roster-final-checks.json').write_text(json.dumps(checks,indent=2)+'\n')
checks.append(run('SelectionRosterPreview',[godot,'--path','.','--script','tests/rendered/SelectionRosterPreview.gd'],240,True))
files=['source/match/hud/SelectionRoster.gd','source/match/hud/UnitMenus.gd','tests/headless/SelectionRosterRegression.gd','tests/rendered/SelectionRosterPreview.gd']
checks.append(run('format',['/tmp/horizon-gdtools/bin/gdformat','--check',*files]))
checks.append(run('lint',['/tmp/horizon-gdtools/bin/gdlint',*files]))
checks.append(run('whitespace',['git','diff','--check']))
(base/'selection-roster-final-checks.json').write_text(json.dumps(checks,indent=2)+'\n')
raise SystemExit(1 if any(r['exit'] or r['script_error'] or r['load_error'] or r['pass_marker'] is False for r in checks) else 0)
