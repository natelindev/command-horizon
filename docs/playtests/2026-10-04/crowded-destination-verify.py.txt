import concurrent.futures,json,subprocess,time
from pathlib import Path
base=Path('docs/playtests/2026-10-04');godot='./tmp/tooling/godot-4.6/Godot.app/Contents/MacOS/Godot'
checks=[]
def run(name,args,marker=True,limit=300):
 log=base/('crowded-destination-final-'+name+'.log');start=time.monotonic()
 with log.open('w') as out:r=subprocess.run(args,stdout=out,stderr=subprocess.STDOUT,timeout=limit)
 text=log.read_text();row={'name':name,'exit':r.returncode,'script_error':'SCRIPT ERROR' in text,'load_error':'Failed to load' in text,'pass_marker':('0 failures' in text or 'PASS:' in text or '0 unresolved checks' in text) if marker else None,'wall_seconds':round(time.monotonic()-start,1),'log':str(log)};print(json.dumps(row),flush=True);return row
suites=['CrowdedDestinationRegression','DirectOrdersRegression','OrdersRegression','NavigationRegression','SaveRestoreRegression','SaveRestoreCombat','ContactApproachRegression','HarvestContinuity','PolicyAdvanceRegression','PolicyRecoveryRegression','SupplyLineRegression','SkirmishCoordinationRegression']
with concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:checks=list(pool.map(lambda n:run(n,[godot,'--headless','--path','.','--fixed-fps','60','--script','tests/headless/'+n+'.gd']),suites))
(base/'crowded-destination-shared-checks.json').write_text(json.dumps(checks,indent=2)+'\n')
def match(case):
 name='ResponsiveBastion' if case==1 else 'ResponsiveHelix'
 return run(name,[godot,'--headless','--path','.','--fixed-fps','60','--script','tmp/MixedResponsiveMatches.gd','--','--case='+str(case),'--repeats=1','--ground-first','--proactive','--output=res://'+str(base/('crowded-destination-final-'+name+'.json')),'--archive=res://'+str(base/('crowded-destination-final-'+name+'-stall.json'))],True,1200)
with concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:checks.extend(pool.map(match,[1,5]))
checks.append(run('import',[godot,'--headless','--path','.','--editor','--quit'],False))
files=['source/match/units/actions/Moving.gd','tests/headless/CrowdedDestinationRegression.gd','tests/rendered/CrowdedDestinationPreview.gd','tests/headless/SkirmishDefensePolicy.gd','tests/headless/PolicyAdvanceRegression.gd','tmp/MixedOperationsMatches.gd','tmp/MixedStagedMatches.gd','tmp/MixedAllocationMatches.gd','tmp/MixedResponsiveMatches.gd']
checks.append(run('format',['/tmp/horizon-gdtools/bin/gdformat','--check',*files],False))
checks.append(run('lint',['/tmp/horizon-gdtools/bin/gdlint',*files],False))
checks.append(run('whitespace',['git','diff','--check'],False))
(base/'crowded-destination-final-checks.json').write_text(json.dumps(checks,indent=2)+'\n')
raise SystemExit(1 if any(r['exit'] or r['script_error'] or r['load_error'] or r['pass_marker'] is False for r in checks) else 0)
