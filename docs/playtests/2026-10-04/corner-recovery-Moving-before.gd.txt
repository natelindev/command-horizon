extends "res://source/match/units/actions/Action.gd"

var _target_position = null
var _arrival_elapsed = 0.0
var _arrival_progress = null

@onready var _unit = Utils.NodeEx.find_parent_with_group(self, "units")
@onready var _movement_trait = _unit.find_child("Movement")


static func is_applicable(unit):
	return unit.find_child("Movement") != null


func _init(target_position):
	_target_position = target_position


func _ready():
	if restoring:
		return
	_movement_trait.move(_target_position)
	_movement_trait.movement_finished.connect(_on_movement_finished)


func _process(delta):
	if (
		not _movement_trait._synchronized
		or not _movement_trait._has_destination
		or _unit.movement_domain != Constants.Match.Navigation.Domain.TERRAIN
	):
		return
	var point = _unit.global_position_yless
	if _arrival_progress == null:
		_arrival_progress = point
	_arrival_elapsed += delta
	if _arrival_elapsed < 2.0:
		return
	_arrival_elapsed = 0.0
	var stalled = point.distance_to(_arrival_progress) < 0.2
	_arrival_progress = point
	if stalled and _occupied_endpoint():
		_movement_trait.stop()
		queue_free()


func _occupied_endpoint():
	var path = _movement_trait.get_current_navigation_path()
	if path.is_empty() or _movement_trait.get_current_navigation_path_index() != path.size() - 1:
		return false
	var end = path[-1] * Vector3(1, 0, 1)
	# A plain point order can finish beside a settled ally occupying its final
	# reachable position. MovingToUnit owns its own process/contact checks.
	for other in _unit.player.get_children():
		if (
			other == _unit
			or not other.is_in_group("units")
			or other.is_queued_for_deletion()
			or other.hp <= 0
			or other.movement_speed <= 0
			or other.movement_domain != _unit.movement_domain
		):
			continue
		var movement = other.get_node_or_null("Movement")
		if (
			movement == null
			or movement._has_destination
			or movement.velocity.length_squared() >= 0.01
		):
			continue
		var spacing = _unit.radius + other.radius
		if (
			other.global_position_yless.distance_to(end) < spacing
			and _unit.global_position_yless.distance_to(end) <= spacing + 0.2
		):
			return true
	return false


func _exit_tree():
	if is_inside_tree():
		_movement_trait.stop()


func _on_movement_finished():
	queue_free()
