# gdlint: disable=max-returns
extends RefCounted

const STATE_BYTES = 16 * 1024
const INTENT_KEYS = ["kind", "assignment", "target", "status"]
const KINDS = ["hold", "move", "attack", "repair", "escort"]
const ASSIGNMENTS = ["anchor", "safe", "route_a", "route_b"]
var source = ""
var error = ""
var suspended = true
var metrics = {}
var _native = null
var _state = {}


static func available():
	return ClassDB.class_exists("RtsLuaSandbox")


func compile_source(text):
	if not available():
		return {"ok": false, "error": "Lua runtime is unavailable on this installation."}
	var candidate = ClassDB.instantiate("RtsLuaSandbox")
	var result = candidate.compile_source(text)
	# Invalid edits do not replace a working runtime. The caller controls Apply.
	if not result.ok:
		return result
	_native = candidate
	source = text
	_state = {}
	error = ""
	suspended = false
	metrics = result
	return {"ok": true}


func resume():
	return compile_source(source)


func step(snapshot):
	if suspended or _native == null:
		return {"ok": false, "error": error if not error.is_empty() else "Program is suspended."}
	var result = _native.step(snapshot, _state)
	metrics = result.duplicate()
	metrics.erase("value")
	if not result.ok:
		return _fail(result.error)
	var value = result.value
	if not value is Dictionary or value.size() != 2 or not value.has_all(["intent", "state"]):
		return _fail("Return exactly {intent = {...}, state = {...}}.")
	if not value.state is Dictionary:
		return _fail("Program state must be a record.")
	if JSON.stringify(value.state).to_utf8_buffer().size() > STATE_BYTES:
		return _fail("Program state exceeds 16 KiB.")
	var problem = validate_intent(value.intent)
	if not problem.is_empty():
		return _fail(problem)
	_state = value.state.duplicate(true)
	return {"ok": true, "intent": value.intent.duplicate(true)}


func state_snapshot():
	return _state.duplicate(true)


func checkpoint():
	return {
		"api_version": 1,
		"source_hash": source.sha256_text(),
		"state": state_snapshot(),
		"suspended": suspended,
		"error": error,
	}


func restore_checkpoint(text, saved):
	if (
		not text is String
		or not saved is Dictionary
		or saved.size() != 5
		or not saved.get("api_version") is int
		or saved.get("api_version") != 1
		or saved.get("source_hash") != text.sha256_text()
		or not saved.get("suspended") is bool
		or not saved.get("error") is String
		or saved.error.length() > 2048
	):
		return {"ok": false, "error": "Program checkpoint does not match its source/API."}
	var valid_state = saved.get("state") is Dictionary
	valid_state = valid_state and _checkpoint_value(saved.state, 0, [2048])
	valid_state = valid_state and JSON.stringify(saved.state).to_utf8_buffer().size() <= STATE_BYTES
	# Prepare a fresh bounded VM. Failure leaves this runtime and its state untouched.
	var result = compile_source(text)
	if not result.ok:
		return result
	_state = saved.state.duplicate(true) if valid_state else {}
	suspended = saved.suspended
	error = saved.error
	return {
		"ok": true,
		"warning": "" if valid_state else "Invalid program memory discarded.",
	}


func _checkpoint_value(value, depth, budget):
	budget[0] -= 1
	if depth > 8 or budget[0] < 0:
		return false
	match typeof(value):
		TYPE_NIL, TYPE_BOOL, TYPE_INT:
			return true
		TYPE_FLOAT:
			return is_finite(value)
		TYPE_STRING:
			return value.to_utf8_buffer().size() <= 4096
		TYPE_ARRAY:
			if value.size() > budget[0]:
				return false
			for item in value:
				if not _checkpoint_value(item, depth + 1, budget):
					return false
			return true
		TYPE_DICTIONARY:
			if value.size() > budget[0]:
				return false
			for key in value:
				if not key is String or key.to_utf8_buffer().size() > 4096:
					return false
				if not _checkpoint_value(value[key], depth + 1, budget):
					return false
			return true
	return false


func _fail(message):
	error = message
	suspended = true
	return {"ok": false, "error": error}


static func validate_intent(intent):
	if not intent is Dictionary or not intent.get("kind") in KINDS:
		return "Intent must name hold, move, attack, repair or escort."
	for key in intent:
		if not key in INTENT_KEYS:
			return "Unsupported intent field: %s." % key
	if intent.has("status") and (not intent.status is String or intent.status.length() > 120):
		return "Status must be text of at most 120 characters."
	if intent.kind == "move":
		if not intent.get("assignment") in ASSIGNMENTS or intent.has("target"):
			return "Move must name anchor, safe, route_a or route_b."
	elif intent.has("assignment"):
		return "Only move accepts a point assignment."
	if intent.kind in ["attack", "repair"]:
		if not intent.get("target") is int or intent.target < 1:
			return "Attack/repair must name a contact ID from the current snapshot."
	elif intent.has("target"):
		return "This intent does not accept a target ID."
	return ""
