extends "res://tmp/PresetRecoveryMatches.gd"

var field_application
var field_records
var field_rules
var field_mode = "lua"
var field_recovering = {}
var field_blocked = {}


func _run():
	field_application = load("res://source/match/programs/ProgramApplication.gd")
	field_records = load("res://source/match/programs/ProgramRecord.gd")
	field_rules = load("res://source/match/programs/VisualRules.gd")
	await super()


func _play_case(index):
	field_mode = "lua" if index % 4 == 1 else "visual"
	field_recovering.clear()
	field_blocked.clear()
	await super(index)


func _command():
	super()
	if phase == "attack":
		return
	for unit in _policy_army():
		if (
			unit.program == null
			or unit.program.mode not in ["Ready", "Running"]
			or not unit.get_meta("policy_programmed", false)
			or unit.get_meta("authored_policy_attempted", false)
		):
			continue
		var document = field_rules.starter(1)
		document.parameters = {"retreat": 40, "recover": 85}
		var entry = field_records.visual("Authored field guard", 1, document, [unit.type])
		if field_mode == "lua":
			entry = field_records.lua(
				"Converted field guard", 1, entry.source, [unit.type], ["movement", "attack"]
			)
		unit.set_meta("authored_policy_attempted", true)
		var result = field_application.apply(entry, [unit], human)
		_check(result.ok, "Authored field guard failed to apply: " + str(result))
		if result.ok:
			unit.set_meta("authored_policy", true)
			metrics.authored_applied = metrics.get("authored_applied", 0) + 1


func _measure():
	super()
	metrics.authoring_mode = field_mode
	var seen = {}
	for unit in _policy_army():
		if not unit.get_meta("authored_policy", false):
			continue
		var id = unit.get_instance_id()
		seen[id] = true
		var previous = field_recovering.get(id, false)
		var running = unit.program.mode == "Running"
		var recovering = (
			running and unit.program.lua_runtime.state_snapshot().get("retreating", false)
		)
		if recovering and not previous:
			metrics.authored_recovery_entries = metrics.get("authored_recovery_entries", 0) + 1
		if previous and not recovering:
			var outcome = "returns" if running and unit.hp >= unit.hp_max * .85 else "interrupted"
			var key = "authored_recovery_" + outcome
			metrics[key] = metrics.get(key, 0) + 1
		field_recovering[id] = recovering
		if unit.program.mode == "Blocked" and not field_blocked.has(id):
			field_blocked[id] = true
			if not metrics.has("authored_blocked"):
				metrics.authored_blocked = []
			var archive_id = _capture_stationary_state(unit)
			metrics.authored_blocked.append(
				{
					"seconds": game.simulation_time,
					"role": unit.type,
					"reason": unit.program.reason,
					"archive_unit": archive_id
				}
			)
			print("AUTHORED BLOCKED ", JSON.stringify(metrics.authored_blocked[-1]))
	for id in field_recovering.keys():
		if not seen.has(id):
			if field_recovering[id]:
				metrics.authored_recovery_losses = metrics.get("authored_recovery_losses", 0) + 1
			field_recovering.erase(id)
