import concurrent.futures, hashlib, json, pathlib, subprocess, time
ROOT=pathlib.Path.cwd()
OUT=ROOT/'docs/playtests/2026-10-04'
GODOT=ROOT/'tmp/tooling/godot-4.6/Godot.app/Contents/MacOS/Godot'
def run(case):
    stem=f'regional-final-case-{case}'
    args=[str(GODOT),'--headless','--path',str(ROOT),'--fixed-fps','60','--script','tmp/RegionalSkirmishMatches.gd','--',f'--case={case}','--repeats=1','--ground-first','--proactive',f'--output=res://docs/playtests/2026-10-04/{stem}.json',f'--archive=res://docs/playtests/2026-10-04/{stem}-stall.json']
    start=time.time()
    with (OUT/(stem+'.log')).open('w') as log:
        result=subprocess.run(args,stdout=log,stderr=subprocess.STDOUT,timeout=3600)
    content=(OUT/(stem+'.log')).read_text()
    row={'case':case,'exit_code':result.returncode,'wall_seconds':round(time.time()-start,2),'log':stem+'.log','record':stem+'.json','script_error':any(x in content for x in ['SCRIPT ERROR','Failed to load script','Parse Error']),'terminal_checks':'Skirmish matrix: 1 records, 0 unresolved checks' in content}
    print(json.dumps(row),flush=True)
    return row
with concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:
    rows=list(pool.map(run,range(8)))
(OUT/'regional-final-execution.json').write_text(json.dumps(rows,indent=2)+'\n')
