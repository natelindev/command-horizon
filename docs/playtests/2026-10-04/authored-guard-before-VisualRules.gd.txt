# gdlint: disable=max-returns
extends RefCounted

const VERSION = 1
const MAX_RULES = 12
const STATES = ["normal", "regroup", "support", "patrol"]
const CONDITIONS = {
	"always": "Always",
	"retreating": "Needs recovery",
	"hp_below": "Health below (%)",
	"enemy": "Enemy in weapon range",
	"ally_damaged": "Repairable ally nearby",
	"missing_anchor": "Area is not assigned",
	"away_anchor": "Farther from area (m)",
	"away_safe": "Farther from safe point (m)",
	"near_route_a": "Near route start (m)",
	"near_route_b": "Near route end (m)",
}
const ACTIONS = {
	"hold": "Hold & defend",
	"move_anchor": "Move to area",
	"move_safe": "Move to safe point",
	"attack_nearest": "Attack nearest enemy",
	"attack_weakest": "Attack weakest enemy",
	"repair_weakest": "Repair weakest ally",
	"escort": "Follow assigned ally",
	"patrol": "Patrol assigned route",
	"move_route_a": "Move to route start",
	"move_route_b": "Move to route end",
}
const PARAMETER_SCHEMA = {
	"retreat": {"type": "number", "minimum": 10, "maximum": 60, "step": 5},
	"recover": {"type": "number", "minimum": 40, "maximum": 95, "step": 5},
}
const STARTERS = ["Hold position", "Guard & regroup", "Scout patrol", "Escort ally", "Repair area"]


static func rule(condition = "always", action = "hold", value = 0):
	return {
		"when": condition, "value": value, "action": action, "state": "any", "next_state": "keep"
	}


static func starter(index):
	var rows = [rule()]
	match index:
		1:
			rows = [
				rule("retreating", "move_safe"),
				rule("missing_anchor", "move_anchor"),
				rule("away_anchor", "move_anchor", 2),
				rule("enemy", "attack_nearest")
			]
		2:
			rows = [rule("retreating", "move_safe"), rule("always", "patrol")]
		3:
			rows = [rule("retreating", "move_safe"), rule("always", "escort")]
		4:
			rows = [rule("missing_anchor", "move_anchor"), rule("ally_damaged", "repair_weakest")]
	return {
		"schema": VERSION,
		"parameters": {"retreat": 30, "recover": 80},
		"initial_state": "normal",
		"rules": rows
	}


static func range_for(condition):
	if condition == "hp_below":
		return Vector3(5, 95, 5)
	if condition in ["away_anchor", "away_safe", "near_route_a", "near_route_b"]:
		return Vector3(0.5, 12, 0.5)
	return Vector3.ZERO


static func validate(document):
	if not document is Dictionary or document.size() != 4:
		return "Invalid rule document."
	if document.get("schema") != VERSION or not document.get("initial_state") in STATES:
		return "Unsupported rule version or starting state."
	var parameters = document.get("parameters")
	if not parameters is Dictionary or parameters.size() != 2:
		return "Choose retreat and recovery thresholds."
	for key in PARAMETER_SCHEMA:
		var bounds = PARAMETER_SCHEMA[key]
		if not _number(parameters.get(key), Vector3(bounds.minimum, bounds.maximum, bounds.step)):
			return "Use valid health thresholds in 5% steps."
	if parameters.recover < parameters.retreat + 10:
		return "Recovery must be at least 10 points above retreat."
	var rows = document.get("rules")
	if not rows is Array or rows.is_empty() or rows.size() > MAX_RULES:
		return "Use between 1 and 12 rules."
	for row in rows:
		if not row is Dictionary or row.size() != 5:
			return "Invalid rule fields."
		if not CONDITIONS.has(row.get("when")) or not ACTIONS.has(row.get("action")):
			return "Choose a supported condition and action."
		if (
			not row.get("state") in STATES + ["any"]
			or not row.get("next_state") in STATES + ["keep"]
		):
			return "Choose a supported state."
		if not _number(row.get("value"), range_for(row.when)):
			return "Use a valid condition threshold."
	return ""


static func _number(value, bounds):
	if not value is int and not value is float:
		return false
	if not is_finite(float(value)) or value < bounds.x or value > bounds.y:
		return false
	return bounds.z == 0 or is_equal_approx(float(value) / bounds.z, round(float(value) / bounds.z))


static func requirements(document):
	var result = ["movement"]
	if not document is Dictionary or not document.get("rules") is Array:
		return result
	for row in document.get("rules", []):
		if not row is Dictionary:
			continue
		if row.get("action") in ["attack_nearest", "attack_weakest"] and not "attack" in result:
			result.append("attack")
		if row.get("action") == "repair_weakest" and not "repair" in result:
			result.append("repair")
	# Capability order is canonical regardless of rule priority.
	result.sort()
	return result


static func assignments(document):
	var result = []
	for row in document.rules:
		var needed = []
		if (
			row.action in ["move_anchor", "repair_weakest"]
			or row.when in ["missing_anchor", "away_anchor"]
		):
			needed.append("Area")
		if row.action == "move_safe" or row.when == "away_safe":
			needed.append("Safe point")
		if (
			row.action in ["patrol", "move_route_a", "move_route_b"]
			or row.when in ["near_route_a", "near_route_b"]
		):
			needed.append("Route")
		if row.action == "escort":
			needed.append("Escort ally")
		for name in needed:
			if not name in result:
				result.append(name)
	return result


static func compile(document):
	var problem = validate(document)
	if not problem.is_empty():
		return {"ok": false, "error": problem}
	var lines = [
		"-- Visual rules v1. Edit the rule document; this source is generated.",
		"return function(snapshot, state)",
		"    local function distance(point)",
		"        if not point then return nil end",
		"        local dx = snapshot.self.position.x - point.x",
		"        local dz = snapshot.self.position.z - point.z",
		"        return math.sqrt(dx * dx + dz * dz)",
		"    end",
		"    local health = snapshot.self.hp / snapshot.self.hp_max * 100",
		"    state.mode = state.mode or '%s'" % document.initial_state,
		"    if health < %d then state.retreating = true end" % document.parameters.retreat,
		"    if health >= %d then state.retreating = false end" % document.parameters.recover,
		"    local enemy, weakest, repair = nil, nil, nil",
		"    for _, contact in ipairs(snapshot.enemies) do",
		"        if contact.can_attack then",
		"            if not enemy or contact.distance < enemy.distance then enemy = contact end",
		"            if not weakest or contact.hp / contact.hp_max < weakest.hp / weakest.hp_max then",
		"                weakest = contact",
		"            end",
		"        end",
		"    end",
		"    local function in_area(contact)",
		"        local point = snapshot.assignments.anchor",
		"        if not point then return false end",
		"        local dx = contact.position.x - point.x",
		"        local dz = contact.position.z - point.z",
		"        return dx * dx + dz * dz <= 36",
		"    end",
		"    for _, contact in ipairs(snapshot.allies) do",
		"        if contact.can_repair and contact.hp < contact.hp_max and in_area(contact) then",
		"            if not repair or contact.hp / contact.hp_max < repair.hp / repair.hp_max then",
		"                repair = contact",
		"            end",
		"        end",
		"    end",
		"    local anchor = distance(snapshot.assignments.anchor)",
		"    local safe = distance(snapshot.assignments.safe)",
		"    local route_a = distance(snapshot.assignments.route_a)",
		"    local route_b = distance(snapshot.assignments.route_b)",
	]
	for index in range(document.rules.size()):
		var row = document.rules[index]
		var condition = _condition(row)
		if row.state != "any":
			condition += " and state.mode == '%s'" % row.state
		if row.action in ["attack_nearest", "attack_weakest"]:
			condition += " and enemy ~= nil"
		if row.action == "repair_weakest":
			condition += " and repair ~= nil"
		lines.append("    if " + condition + " then")
		if row.next_state != "keep":
			lines.append("        state.mode = '%s'" % row.next_state)
		var intent = _intent(row.action, lines)
		lines.append(
			(
				"        return {intent = {%s, status = 'Rule %d: %s'}, state = state}"
				% [intent, index + 1, ACTIONS[row.action]]
			)
		)
		lines.append("    end")
	lines.append(
		"    return {intent = {kind = 'hold', status = 'No rule matched: hold'}, state = state}"
	)
	lines.append("end\n")
	return {"ok": true, "source": "\n".join(lines)}


static func _condition(row):
	var number = "%.1f" % row.value
	match row.when:
		"retreating":
			return "state.retreating"
		"hp_below":
			return "health < " + number
		"enemy":
			return "enemy ~= nil"
		"ally_damaged":
			return "repair ~= nil"
		"missing_anchor":
			return "anchor == nil"
		"away_anchor":
			return "anchor ~= nil and anchor > " + number
		"away_safe":
			return "safe ~= nil and safe > " + number
		"near_route_a":
			return "route_a ~= nil and route_a <= " + number
		"near_route_b":
			return "route_b ~= nil and route_b <= " + number
	return "true"


static func _intent(action, lines):
	match action:
		"attack_nearest":
			return "kind = 'attack', target = enemy.id"
		"attack_weakest":
			return "kind = 'attack', target = weakest.id"
		"repair_weakest":
			return "kind = 'repair', target = repair.id"
		"escort":
			return "kind = 'escort'"
		"patrol":
			# Both points are required; missing route never guesses a world destination.
			lines.append("        if not route_a or not route_b then")
			lines.append(
				(
					"            return {intent = {kind = 'move', "
					+ "assignment = not route_a and 'route_a' or 'route_b'}, state = state}"
				)
			)
			lines.append("        end")
			lines.append("        if not state.route_b and route_a <= 1 then state.route_b = true")
			lines.append(
				"        elseif state.route_b and route_b <= 1 then state.route_b = false end"
			)
			return "kind = 'move', assignment = state.route_b and 'route_b' or 'route_a'"
	if action.begins_with("move_"):
		return "kind = 'move', assignment = '%s'" % action.trim_prefix("move_")
	return "kind = 'hold'"
