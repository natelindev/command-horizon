import concurrent.futures,json,subprocess,time
from pathlib import Path
base=Path('docs/playtests/2026-10-04'); godot='./tmp/tooling/godot-4.6/Godot.app/Contents/MacOS/Godot'
names=['ProgramRegression','LuaProgramRegression','AuthoredGuardRegression','AuthoredRecoveryRegression','SaveSnapshotRegression','SaveRestoreRegression','SaveRestoreEdges','SaveRestoreCombat','OrdersRegression']
def run(n):
 p=base/('blocked-assignment-shared-'+n+'.log'); start=time.monotonic()
 with p.open('w') as f: r=subprocess.run([godot,'--headless','--path','.','--fixed-fps','60','--script','tests/headless/'+n+'.gd'],stdout=f,stderr=subprocess.STDOUT,timeout=360)
 t=p.read_text(); d={'name':n,'exit':r.returncode,'script_error':'SCRIPT ERROR' in t,'load_error':'Failed to load' in t,'pass_marker':('0 failures' in t or 'PASS:' in t),'wall_seconds':round(time.monotonic()-start,1),'log':str(p)};print(json.dumps(d),flush=True);return d
with concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool: rows=list(pool.map(run,names))
(base/'blocked-assignment-shared-checks.json').write_text(json.dumps(rows,indent=2)+'\n')
raise SystemExit(int(any(r['exit'] or r['script_error'] or r['load_error'] or not r['pass_marker'] for r in rows)))
