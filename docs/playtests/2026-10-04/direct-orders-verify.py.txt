import concurrent.futures,json,subprocess,time
from pathlib import Path
base=Path('docs/playtests/2026-10-04')
godot='./tmp/tooling/godot-4.6/Godot.app/Contents/MacOS/Godot'
checks=[]
def run(name,args,limit=240,marker=False):
 log=base/('direct-orders-final-'+name+'.log')
 start=time.monotonic()
 with log.open('w') as out:
  result=subprocess.run(args,stdout=out,stderr=subprocess.STDOUT,timeout=limit)
 content=log.read_text()
 row={'suite':name,'exit':result.returncode,'script_error':'SCRIPT ERROR' in content,'load_error':'Failed to load' in content,'pass_marker':('0 failures' in content or 'PASS:' in content) if marker else None,'wall_seconds':round(time.monotonic()-start,1),'log':str(log)}
 print(json.dumps(row),flush=True)
 return row
def save():
 (base/'direct-orders-final-checks.json').write_text(json.dumps(checks,indent=2)+'\n')
checks.append(run('import',[godot,'--headless','--path','.','--editor','--quit']))
suites=['DirectOrdersRegression','KeyBindingsRegression','OrdersRegression','ProgramRegression','SelectionRosterRegression','SaveRestoreRegression','SaveRestoreCombat','CampaignControlsRegression','SubtitleLayoutRegression','BattleAlertsRegression','BattlePlanRegression','CampaignRegression']
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
 futures=[pool.submit(run,name,[godot,'--headless','--path','.','--fixed-fps','60','--script','tests/headless/'+name+'.gd'],300,True) for name in suites]
 for future in concurrent.futures.as_completed(futures):
  checks.append(future.result())
  save()
for name in ['DirectOrdersPreview','KeyBindingsPreview','CampaignScalePreview','SelectionRosterPreview']:
 checks.append(run(name,[godot,'--path','.','--script','tests/rendered/'+name+'.gd'],300,True))
 save()
files=['source/match/hud/DirectOrders.gd','source/match/hud/SkirmishHUD.gd','source/match/hud/UnitMenus.gd','source/match/hud/BattleAlertsPanel.gd','source/match/hud/SkirmishPlanPanel.gd','source/input/KeyBindings.gd','tests/headless/DirectOrdersRegression.gd','tests/headless/KeyBindingsRegression.gd','tests/rendered/DirectOrdersPreview.gd']
checks.append(run('format',['/tmp/horizon-gdtools/bin/gdformat','--check',*files]))
checks.append(run('lint',['/tmp/horizon-gdtools/bin/gdlint',*files]))
checks.append(run('whitespace',['git','diff','--check']))
save()
raise SystemExit(1 if any(row['exit'] or row['script_error'] or row['load_error'] or row['pass_marker'] is False for row in checks) else 0)
