extends "res://tests/headless/SaveRestoreEdges.gd"


func _run():
	root.get_node("SaveManager").autosaves_enabled = false
	for trial in 2:
		var restore = Restore.new()
		var result = await restore.restore(
			self,
			FileStore.new().read_file("res://docs/playtests/2026-10-04/guard-travel-after-3-0.json")
		)
		_check(result.ok, "Corner restore failed")
		if not result.ok:
			continue
		game = result.game
		for player in get_nodes_in_group("players"):
			player.set_process(false)
		var unit = restore.entities["unit-13"]
		var movement = unit.get_node("Movement")
		await _frames(30)
		var start = unit.position
		var map = movement.get_navigation_map()
		for angle in 8:
			var direction = Vector3(cos(angle * TAU / 8), 0, sin(angle * TAU / 8))
			var candidate = start + direction * 1.8
			var projected = NavigationServer3D.map_get_closest_point(map, candidate)
			projected.y = 0
			print(
				"CORNER CLEARANCE ",
				angle,
				" candidate=",
				candidate,
				" projected=",
				projected,
				" offset=",
				projected.distance_to(candidate)
			)
		unit.action._stalled = 0.0
		var previous_recovery = -1.0
		paused = false
		for step in 40:
			for frame in 30:
				if trial == 1:
					# Diagnostic: drive the requested recovery strictly east instead of
					# blending back toward the obstructed west/south path.
					var route = movement.get_current_navigation_path()
					if movement._has_destination and not route.is_empty():
						var index = mini(
							movement.get_current_navigation_path_index(), route.size() - 1
						)
						var direction = (
							((route[index] - unit.position) * Vector3(1, 0, 1)).normalized()
						)
						movement._recovery_direction = Vector3.RIGHT - direction * .35
						movement._recovery_left = .1
				await physics_frame
			print(
				"CORNER ",
				JSON.stringify(
					{
						"trial": trial,
						"time": game.simulation_time,
						"position": str(unit.position),
						"state": unit.program.mode,
						"reason": unit.program.reason,
						"stalled": unit.action._stalled,
						"path": str(movement.get_current_navigation_path()),
						"index": movement.get_current_navigation_path_index(),
						"requested": str(movement.velocity),
						"progress": unit.position.distance_to(start)
					}
				)
			)
		_check(unit.program.mode == "Running", "Corner guard blocked with live traffic")
		game.queue_free()
		await _frames(5)
	for error in errors:
		push_error(error)
	print("Guard corner: %d failures" % errors.size())
	quit(0 if errors.is_empty() else 1)
