import concurrent.futures, subprocess, pathlib, json
out=pathlib.Path('docs/playtests/2026-10-04')
names=['SkirmishCoordinationRegression','SkirmishCoordinationFireRegression','SkirmishCoordinationMetricsRegression','SkirmishDefenseRegression','SkirmishRecoveryRegression','SkirmishOffenseRegression','SkirmishSupportRegression','SaveRestoreCombat','OrdersRegression']
def run(name):
 path=out/('coordination-handoff-check-'+name+'.log')
 args=['tmp/tooling/godot-4.6/Godot.app/Contents/MacOS/Godot','--headless','--path','.','--fixed-fps','60','--script','tests/headless/'+name+'.gd']
 with path.open('w') as log:r=subprocess.run(args,stdout=log,stderr=subprocess.STDOUT,timeout=600)
 text=path.read_text();row={'test':name,'exit':r.returncode,'script_error':'SCRIPT ERROR' in text};print(json.dumps(row),flush=True);return row
with concurrent.futures.ThreadPoolExecutor(max_workers=2) as pool:rows=list(pool.map(run,names))
(out/'coordination-handoff-checks.json').write_text(json.dumps(rows,indent=2)+'\n')
