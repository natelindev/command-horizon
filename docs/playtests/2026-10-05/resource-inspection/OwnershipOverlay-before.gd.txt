extends Control

const Ownership = preload("res://source/match/hud/Ownership.gd")
var game
var _pointer = null


func _ready():
	name = "OwnershipOverlay"
	game = find_parent("Match")
	process_mode = Node.PROCESS_MODE_ALWAYS
	mouse_filter = Control.MOUSE_FILTER_IGNORE
	set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)


func _notification(what):
	if what == NOTIFICATION_DISABLED:
		_pointer = null


func _process(_delta):
	queue_redraw()


func _input(event):
	if event is InputEventMouseMotion:
		_pointer = event.position


func _hovered_unit(camera):
	if _pointer == null or get_viewport().gui_get_hovered_control() != null:
		return null
	var origin = camera.project_ray_origin(_pointer)
	var query = PhysicsRayQueryParameters3D.create(
		origin, origin + camera.project_ray_normal(_pointer) * 1000, 2
	)
	query.collide_with_areas = true
	query.collide_with_bodies = false
	var hit = camera.get_world_3d().direct_space_state.intersect_ray(query)
	return hit.get("collider")


func markers():
	var result = []
	if not Globals.options.ownership_symbols or not is_instance_valid(game):
		return result
	var camera = get_viewport().get_camera_3d()
	if camera == null:
		return result
	var hovered = _hovered_unit(camera)
	for unit in get_tree().get_nodes_in_group("units"):
		if not Ownership.known_now(unit, game):
			continue
		var health = unit.get_node_or_null("HealthBar")
		var position_3d = (
			health.global_position if health != null else unit.global_position + Vector3.UP * 2
		)
		if camera.is_position_behind(position_3d):
			continue
		var point = camera.unproject_position(position_3d) + Vector2(0, -12)
		if not get_viewport_rect().grow(-8).has_point(point):
			continue
		result.append(
			{
				"unit": unit,
				"point": point,
				"kind": Ownership.kind(unit, game),
				"caption": Ownership.caption(unit, game),
				"hovered": unit == hovered
			}
		)
	return result


func _draw():
	var hovered = null
	for marker in markers():
		Ownership.paint(self, marker.point, marker.kind, 4.5, Color("eef1f1"))
		if marker.hovered:
			hovered = marker
	if hovered == null:
		return
	var font = ThemeDB.fallback_font
	var text_size = font.get_string_size(hovered.caption, HORIZONTAL_ALIGNMENT_LEFT, -1, 15)
	var rect = caption_rect(hovered.point, text_size + Vector2(16, 8))
	if not rect.has_area():
		return
	var background = StyleBoxFlat.new()
	background.bg_color = Color("17202bf2")
	background.set_corner_radius_all(4)
	draw_style_box(background, rect)
	draw_string(
		font,
		rect.position + Vector2(8, font.get_ascent(15) + 4),
		hovered.caption,
		HORIZONTAL_ALIGNMENT_LEFT,
		-1,
		15,
		Color("eef1f1")
	)


func caption_rect(point, dimensions):
	var occupied = []
	for node_name in ["UnitMenus", "Minimap", "Resources"]:
		var control = game.find_child(node_name)
		if control is Control and control.is_visible_in_tree():
			occupied.append(control.get_global_rect().grow(6))
	if "hud" in game and is_instance_valid(game.hud):
		for control in game.hud.get_children():
			if control is Control and control.is_visible_in_tree():
				occupied.append(control.get_global_rect().grow(6))
	if game.subtitles != null and game.subtitles._card.is_visible_in_tree():
		occupied.append(game.subtitles._card.get_global_rect().grow(6))
	var viewport = get_viewport_rect().grow(-4)
	for y in [point.y - dimensions.y - 10, point.y + 12]:
		for x in [point.x - dimensions.x / 2, point.x - dimensions.x, point.x]:
			var rect = Rect2(
				Vector2(clampf(x, 4, maxf(4, viewport.end.x - dimensions.x)), y), dimensions
			)
			if (
				viewport.encloses(rect)
				and not occupied.any(func(block): return block.intersects(rect))
			):
				return rect
	return Rect2()
