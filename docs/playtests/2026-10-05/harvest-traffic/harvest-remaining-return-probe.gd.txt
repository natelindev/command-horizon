extends "res://tests/headless/HarvestTrafficRegression.gd"

var tick = 0
var start = -1.0

func _run():
	physics_frame.connect(_trace_tick)
	await super()

func _trace_tick():
	if game == null or paused:
		return
	if start < 0:
		start = game.simulation_time
	tick += 1
	if tick % 30 != 0:
		return
	for worker in get_nodes_in_group("units"):
		if worker.type != "Worker" or worker.process_mode == Node.PROCESS_MODE_DISABLED:
			continue
		var m = worker.get_node("Movement")
		var path = m.get_current_navigation_path()
		var idx = m.get_current_navigation_path_index()
		var sub = worker.action._sub_action if worker.action != null and "_sub_action" in worker.action else null
		print("RETURN TRACE ",JSON.stringify({"t":game.simulation_time-start,"i":worker.get_index(),"p":str(worker.position),"action":str(worker.action),"cargo":worker.resource_b,"next":str(path[idx]) if idx<path.size() else "none","recovery":m._recovery_left,"rev":m._command_revision,"contact":sub._contact_elapsed if sub != null and "_contact_elapsed" in sub else null,"orbit":sub._orbit_windows if sub != null and "_orbit_windows" in sub else null}))
