from pathlib import Path
stage=Path('/tmp/godot-open-rts-polish-stall-prewarm-prototype');prior=Path('/tmp/godot-open-rts-polish-stall-attribution')
project=(prior/'project.godot').read_text().replace('OpenRTSPolishMacStallAttribution-20261024-1','OpenRTSPolishMacPrewarmPrototype-20261024-1')
(stage/'project.godot').write_text(project)
(stage/'BenchmarkBootstrap.tscn').write_text((prior/'BenchmarkBootstrap.tscn').read_text())
wrapper=(prior/'ReleaseBattleBenchmark.gd').read_text()
wrapper+='''

func _skirmish(faction, map_path = "res://source/match/maps/BastionDesert.tscn", advanced = true):
	var cover = CanvasLayer.new()
	cover.layer = 128
	cover.process_mode = Node.PROCESS_MODE_ALWAYS
	root.add_child(cover)
	var panel = ColorRect.new()
	panel.color = Color("10151b")
	panel.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
	cover.add_child(panel)
	var label = Label.new()
	label.text = "Preparing battlefield"
	label.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	label.vertical_alignment = VERTICAL_ALIGNMENT_CENTER
	label.add_theme_font_size_override("font_size", 24)
	label.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
	cover.add_child(label)
	await super(faction, map_path, advanced)
	await _prewarm_combat_variants()
	cover.queue_free()


func _live_cosmetic_state():
	var fx = game.combat_effects
	var states = {
		"simulation_time": game.simulation_time,
		"combat_rng": fx._random.state,
		"aftermath_rng": fx.aftermath._random.state,
		"projectile_rng": game.projectile_system._random.state,
		"shots": fx.shots_emitted,
		"impacts": fx.impacts_emitted,
		"deaths": fx.deaths_emitted,
		"tracers": fx._tracers.size(),
		"free_tracers": fx._free_tracers.size(),
		"wrecks": fx._wrecks.size(),
		"cook_offs": fx._cook_offs.size(),
		"hit_flashes": fx._hit_flashes.size(),
		"free_hit_flashes": fx._free_hit_flashes.size(),
		"shockwaves": fx._shockwaves.size(),
		"rounds": game.projectile_system._rounds.size(),
		"rounds_fired": game.projectile_system.rounds_fired,
		"unit_rng": {},
	}
	for unit in get_nodes_in_group("units"):
		var visuals = unit.get_node_or_null("BastionVisuals")
		if visuals != null:
			states.unit_rng[str(unit.get_instance_id())] = visuals.motion.random.state
	return states


func _prewarm_combat_variants():
	var paused_before = paused
	paused = true
	await RenderingServer.frame_post_draw
	await RenderingServer.frame_post_draw
	var snapshot_before = Snapshot.new().capture(game)
	var cosmetic_before = _live_cosmetic_state()
	var before_nodes = get_node_count()
	var before_pipelines = _pipeline_counts()
	var started = Time.get_ticks_usec()
	print("PREWARM_BEGIN ", JSON.stringify({"unix_ms": int(Time.get_unix_time_from_system() * 1000), "pipelines": before_pipelines}))
	var holders = []
	var camera = game.get_node("IsometricCamera3D")
	var point = camera._calculate_pivot_point()
	point.y = game.map.height_at(Vector2(point.x, point.z)) + 1.0

	# Existing factories create only cosmetic resources. No gameplay shot, impact,
	# death, scar history or live unit motion RNG is invoked.
	var spare = load("res://source/match/effects/CombatEffects.gd").new()
	game.add_child(spare)
	spare.set_process(false)
	spare.set_physics_process(false)
	spare._camera = null
	holders.append(spare)
	for kind in spare._emitters:
		var emitter = spare._emitters[kind]
		emitter.process_mode = Node.PROCESS_MODE_ALWAYS
		emitter.speed_scale = .01
		emitter.emit_particle(Transform3D(Basis.IDENTITY, point), Vector3.UP, Color.WHITE, Color(point.y, 0, 0, 0), 1 | 2 | 4 | 8 | 16)
	var tracer = spare._take_tracer()
	tracer.node.material_override = spare._glowing_material(Color("b6d9e1"))
	tracer.node.global_position = point
	spare._shockwave(point, 2.0)
	spare._shockwaves[0].node.material_override.set_shader_parameter("progress", .35)
	spare.aftermath.burst(point, 2.0, false)
	for entry in spare.aftermath.debris:
		entry.node.freeze = true

	# Reuse already-loaded visual meshes, deduplicated by their vertex layouts.
	# A new standard override has precisely the flags used by _wreck().
	var geometry = Node3D.new()
	game.add_child(geometry)
	holders.append(geometry)
	var formats = {}
	var wreck_material = StandardMaterial3D.new()
	wreck_material.albedo_color = Color("36383a")
	wreck_material.roughness = .95
	for unit in get_nodes_in_group("units"):
		for source in unit.get_node("Geometry").find_children("*", "MeshInstance3D", true, false):
			if source.mesh == null:
				continue
			var layout = source.mesh.get_class()
			if source.mesh is ArrayMesh:
				for surface in source.mesh.get_surface_count():
					layout += "/" + str(source.mesh.surface_get_format(surface))
			if formats.has(layout):
				continue
			formats[layout] = true
			for wreck in [false, true]:
				var proxy = MeshInstance3D.new()
				proxy.mesh = source.mesh
				proxy.material_override = wreck_material if wreck else source.material_override
				proxy.cast_shadow = source.cast_shadow
				geometry.add_child(proxy)
				proxy.global_transform = Transform3D(source.global_basis, point + Vector3(formats.size() % 3 - 1, 0, formats.size() / 3 - 1))
	var hit = load("res://source/match/ObjectMaterials.gd").hit_flash(geometry, Color("ffd6a1"))

	# Exercise the exact pooled projectile draw variants without allocating a round.
	var renderer = load("res://source/match/effects/ProjectileRenderer.gd").new()
	game.add_child(renderer)
	holders.append(renderer)
	for pool in renderer._pools.values():
		pool.node.multimesh.visible_instance_count = 1
		pool.node.multimesh.set_instance_transform(0, Transform3D(Basis.IDENTITY.scaled(Vector3(.1, .1, .5)), point))
		pool.node.multimesh.set_instance_color(0, Color.WHITE)

	# First scar lazily creates both texture-blit and terrain-overlay variants.
	# Redirect the disposable wear instance to a shared-mesh proxy before painting.
	var terrain = game.map.get_node("Geometry/Terrain")
	var terrain_proxy = MeshInstance3D.new()
	terrain_proxy.mesh = terrain.mesh
	terrain_proxy.material_override = terrain.material_override
	game.add_child(terrain_proxy)
	terrain_proxy.global_transform = terrain.global_transform
	holders.append(terrain_proxy)
	var wear = load("res://source/match/effects/BattlefieldWear.gd").new()
	game.add_child(wear)
	wear.set_process(false)
	wear._create_texture()
	wear._terrain = terrain_proxy
	wear.add_mark(Vector2(point.x, point.z), 1.0, 0.0, 0)
	wear.add_mark(Vector2(point.x, point.z), 1.0, 0.0, 1)
	wear._paint()
	holders.append(wear)

	var previous = _pipeline_counts()
	var stable = 0
	var draws = 0
	var events = []
	while draws < 12 and stable < 3:
		await RenderingServer.frame_post_draw
		draws += 1
		var counts = _pipeline_counts()
		stable = stable + 1 if counts == previous else 0
		events.append({"draw": draws, "ticks_us": Time.get_ticks_usec(), "pipelines": counts})
		previous = counts
	load("res://source/match/ObjectMaterials.gd").clear_hit_flash(hit)
	for node in holders:
		node.queue_free()
	await process_frame
	await process_frame
	await RenderingServer.frame_post_draw
	var snapshot_after = Snapshot.new().capture(game)
	var cosmetic_after = _live_cosmetic_state()
	var snapshot_equal = snapshot_before != null and snapshot_before == snapshot_after
	var cosmetic_equal = cosmetic_before == cosmetic_after
	_check(snapshot_equal, "Prewarm changed paused save snapshot")
	_check(cosmetic_equal, "Prewarm changed live effect counters or RNG")
	_check(stable >= 3, "Prewarm pipelines did not stabilize within twelve draws")
	print("PREWARM_END ", JSON.stringify({"unix_ms": int(Time.get_unix_time_from_system() * 1000), "elapsed_ms": (Time.get_ticks_usec() - started) / 1000.0, "draws": draws, "stable_draws": stable, "mesh_layout_count": formats.size(), "pipelines_before": before_pipelines, "pipelines_after": _pipeline_counts(), "events": events, "snapshot_equal": snapshot_equal, "cosmetic_equal": cosmetic_equal, "live_state": cosmetic_after, "node_count_before": before_nodes, "node_count_after": get_node_count()}))
	paused = paused_before
	_check(paused == paused_before, "Prewarm did not restore pause state")
'''
(stage/'ReleaseBattleBenchmark.gd').write_text(wrapper)
print(stage)
