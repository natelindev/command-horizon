extends "res://tests/headless/UnitInfoTextRegression.gd"


func _run():
	var preferences = root.get_node("Globals").options
	var original = preferences.capture()
	preferences.restore({"screen": 1, "language": "en"})
	await _info_fixture()
	print("LIVE INPUT idle_handler=", game.find_child("IdleWorkerHandler", true, false))
	print("LIVE INPUT idle_badge=", game.find_child("IdleWorkerBadge", true, false))
	paused = false
	await _frames(2)
	_check(not diagnostics.visible, "Diagnostics must begin off")
	await _toggle_info()
	_check(diagnostics.visible, "Queued F1 did not enable diagnostics during unpaused battle")
	print("LIVE INPUT first_F1 diagnostics=", diagnostics.visible, " paused=", paused)
	await _toggle_info()
	_check(not diagnostics.visible, "Queued F1 did not disable diagnostics during unpaused battle")
	print("LIVE INPUT second_F1 diagnostics=", diagnostics.visible, " paused=", paused)
	paused = true
	await _toggle_info()
	_check(diagnostics.visible, "Queued F1 did not enable diagnostics during paused battle")
	await _toggle_info()
	_check(not diagnostics.visible, "Queued F1 did not disable diagnostics during paused battle")
	commander.free()
	paused = false
	game.queue_free()
	await _frames(5)
	preferences.restore(original)
	for message in errors:
		push_error(message)
	print("Live unit info input probe: %d failures" % errors.size())
	await root.get_node("Globals").quit_game(0 if errors.is_empty() else 1)
