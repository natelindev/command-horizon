class_name ReleaseBattleBenchmark
extends "res://tests/rendered/BattleFrameBenchmark.gd"

var _focus_phase = "waiting"
var _focus_exit_count = 0
var _native_focus_token = ""
var _trace_rows = []
var _trace_previous_at = 0
var _trace_previous_pipelines = {}


func _run():
	if is_instance_valid(current_scene):
		current_scene.free()
		current_scene = null
	for arg in OS.get_cmdline_user_args():
		if arg.begins_with("--native-focus-token="):
			_native_focus_token = arg.trim_prefix("--native-focus-token=")
	root.title = "Open RTS Mac Benchmark"
	root.mode = Window.MODE_WINDOWED
	root.size = Vector2i(1920, 1080)
	root.focus_entered.connect(_record_focus.bind(true))
	root.focus_exited.connect(_record_focus.bind(false))
	if "--await-focus" in OS.get_cmdline_user_args():
		var label = Label.new()
		label.text = "Click this window to start the performance check"
		label.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
		label.vertical_alignment = VERTICAL_ALIGNMENT_CENTER
		label.add_theme_font_size_override("font_size", 32)
		root.add_child(label)
		label.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
		print(
			"FOCUS_GATE_READY pid=",
			OS.get_process_id(),
			" title=",
			root.title,
			" focused=",
			root.has_focus()
		)
		while (
			not root.has_focus()
			or (_native_focus_token != "" and not FileAccess.file_exists(_native_focus_token))
		):
			await process_frame
		label.free()
	_focus_phase = "setup"
	print("FOCUS_GATE_START pid=", OS.get_process_id(), " focused=", root.has_focus())
	await super()


func _record_focus(focused):
	if not focused:
		_focus_exit_count += 1
	print(
		"FOCUS_EVENT phase=",
		_focus_phase,
		" focused=",
		focused,
		" ticks_us=",
		Time.get_ticks_usec()
	)


func _measure_frames(label, before_sample = Callable()):
	_focus_phase = label
	var exits_before = _focus_exit_count
	var began_focused = root.has_focus()
	print(
		"FOCUS_PHASE_BEGIN phase=",
		label,
		" focused=",
		began_focused,
		" unix_ms=",
		int(Time.get_unix_time_from_system() * 1000)
	)
	_check(began_focused, "Benchmark phase began without window focus")
	_trace_begin(label + "/warmup")
	# Update ticks can run without a draw (e.g. native window transitions).
	# Warm both the renderer and real-time simulation before sampling draws.
	var warm_until = Time.get_ticks_msec() + 3000
	var warm_samples = []
	var warm_last = Time.get_ticks_usec()
	if before_sample.is_valid():
		before_sample.call()
	for frame in 120:
		await RenderingServer.frame_post_draw
		var now = Time.get_ticks_usec()
		warm_samples.append((now - warm_last) / 1000.0)
		_trace_frame(now)
		warm_last = now
	while Time.get_ticks_msec() < warm_until:
		await RenderingServer.frame_post_draw
		var now = Time.get_ticks_usec()
		warm_samples.append((now - warm_last) / 1000.0)
		_trace_frame(now)
		warm_last = now
	_trace_end(label + "/warmup")
	warm_samples.sort()
	_check(
		root.get_texture().get_size() == Vector2(1920, 1080),
		"Ordinary benchmark raster must remain 1920x1080"
	)
	print(
		"WARMUP ",
		label,
		" frames=",
		warm_samples.size(),
		" p95_ms=",
		snappedf(warm_samples[int(warm_samples.size() * .95)], .01),
		" max_ms=",
		snappedf(warm_samples[-1], .01),
		" frames_over_100ms=",
		warm_samples.filter(func(ms): return ms > 100.0).size()
	)
	_trace_begin(label + "/steady")
	var times = []
	var gpu_ms = 0.0
	var render_cpu_ms = 0.0
	var last = Time.get_ticks_usec()
	var started = game.simulation_time
	var drawn = Engine.get_frames_drawn()
	var physics_started = Engine.get_physics_frames()
	for frame in 240:
		await RenderingServer.frame_post_draw
		var now = Time.get_ticks_usec()
		times.append((now - last) / 1000.0)
		_trace_frame(now)
		gpu_ms += RenderingServer.viewport_get_measured_render_time_gpu(root.get_viewport_rid())
		render_cpu_ms += RenderingServer.viewport_get_measured_render_time_cpu(
			root.get_viewport_rid()
		)
		last = now
	_trace_end(label + "/steady")
	times.sort()
	var mean = times.reduce(func(total, sample): return total + sample, 0.0) / times.size()
	_check(
		Engine.get_frames_drawn() - drawn == times.size(),
		"Benchmark window did not render every sample"
	)
	# A fast machine can render all samples in under a second.
	_check(
		(
			absf((game.simulation_time - started) - mean * times.size() / 1000.0) < 0.15
			and Engine.get_physics_frames() > physics_started
		),
		"Benchmark simulation did not advance in real time"
	)
	print(
		"GAMEPLAY ",
		label,
		" units=",
		get_nodes_in_group("units").size(),
		" mean_ms=",
		snappedf(mean, .01),
		" p95_ms=",
		snappedf(times[int(times.size() * .95)], .01),
		" fps=",
		snappedf(1000.0 / mean, .1),
		" simulation_seconds=",
		game.simulation_time - started,
		" wall_seconds=",
		mean * times.size() / 1000.0,
		" viewport=",
		root.get_visible_rect().size,
		" pixels=",
		root.get_texture().get_size(),
		" gpu_ms=",
		snappedf(gpu_ms / times.size(), .01) if gpu_ms > 0.0 else "unavailable",
		" render_cpu_ms=",
		snappedf(render_cpu_ms / times.size(), .01),
		" adapter=",
		RenderingServer.get_video_adapter_name(),
		" rendered_frames=",
		Engine.get_frames_drawn() - drawn,
		" physics_ticks=",
		Engine.get_physics_frames() - physics_started,
		" paused=",
		paused,
		" draw_calls=",
		RenderingServer.get_rendering_info(
			RenderingServer.RENDERING_INFO_TOTAL_DRAW_CALLS_IN_FRAME
		),
		" primitives=",
		RenderingServer.get_rendering_info(RenderingServer.RENDERING_INFO_TOTAL_PRIMITIVES_IN_FRAME)
	)

	var ended_focused = root.has_focus()
	var exits_during = _focus_exit_count - exits_before
	var valid = began_focused and ended_focused and exits_during == 0
	print(
		"FOCUS_PHASE_END phase=",
		label,
		" focused=",
		ended_focused,
		" lost_focus_events=",
		exits_during,
		" focus_valid=",
		valid,
		" unix_ms=",
		int(Time.get_unix_time_from_system() * 1000)
	)
	_check(valid, "Benchmark phase lost window focus")
	_focus_phase = "between"


func _pipeline_counts():
	return {
		"canvas":
		RenderingServer.get_rendering_info(
			RenderingServer.RENDERING_INFO_PIPELINE_COMPILATIONS_CANVAS
		),
		"mesh":
		RenderingServer.get_rendering_info(
			RenderingServer.RENDERING_INFO_PIPELINE_COMPILATIONS_MESH
		),
		"surface":
		RenderingServer.get_rendering_info(
			RenderingServer.RENDERING_INFO_PIPELINE_COMPILATIONS_SURFACE
		),
		"draw":
		RenderingServer.get_rendering_info(
			RenderingServer.RENDERING_INFO_PIPELINE_COMPILATIONS_DRAW
		),
		"specialization":
		RenderingServer.get_rendering_info(
			RenderingServer.RENDERING_INFO_PIPELINE_COMPILATIONS_SPECIALIZATION
		),
	}


func _trace_begin(label):
	_trace_rows.clear()
	_trace_previous_at = Time.get_ticks_usec()
	_trace_previous_pipelines = _pipeline_counts()
	print(
		"TRACE_PHASE_BEGIN ",
		JSON.stringify(
			{
				"phase": label,
				"ticks_us": _trace_previous_at,
				"unix_ms": int(Time.get_unix_time_from_system() * 1000),
				"pipelines": _trace_previous_pipelines
			}
		)
	)


func _trace_frame(now):
	var counts = _pipeline_counts()
	_trace_rows.append(
		{
			"ticks_us": now,
			"unix_ms": int(Time.get_unix_time_from_system() * 1000),
			"frame_ms": (now - _trace_previous_at) / 1000.0,
			"pipeline_before": _trace_previous_pipelines,
			"pipeline_after": counts,
			"render_cpu_ms":
			RenderingServer.viewport_get_measured_render_time_cpu(root.get_viewport_rid()),
			"physics_ms": Performance.get_monitor(Performance.TIME_PHYSICS_PROCESS) * 1000.0,
			"simulation_time": game.simulation_time
		}
	)
	_trace_previous_at = now
	_trace_previous_pipelines = counts


func _trace_end(label):
	print(
		"TRACE_PHASE_END ",
		JSON.stringify(
			{
				"phase": label,
				"ticks_us": Time.get_ticks_usec(),
				"unix_ms": int(Time.get_unix_time_from_system() * 1000),
				"pipelines": _pipeline_counts(),
				"frames": _trace_rows
			}
		)
	)


func _skirmish(faction, map_path = "res://source/match/maps/BastionDesert.tscn", advanced = true):
	await super(faction, map_path, advanced)
	var paused_before = paused
	paused = true
	var saved_before = Snapshot.new().capture(game)
	var nodes_before = get_node_count()
	var started = Time.get_ticks_usec()
	print(
		"PREWARM_BEGIN ",
		JSON.stringify(
			{
				"unix_ms": int(Time.get_unix_time_from_system() * 1000),
				"pipelines": _pipeline_counts()
			}
		)
	)
	var prepared = await load("res://source/match/effects/CombatPrewarm.gd").prepare(game)
	var elapsed_ms = (Time.get_ticks_usec() - started) / 1000.0
	var snapshot_equal = saved_before != null and Snapshot.new().capture(game) == saved_before
	_check(prepared, "Production prewarm did not complete")
	_check(snapshot_equal, "Production prewarm changed paused snapshot")
	_check(get_node_count() == nodes_before, "Production prewarm retained nodes")
	_check(paused, "Production prewarm resumed caller")
	print(
		"PREWARM_END ",
		JSON.stringify(
			{
				"unix_ms": int(Time.get_unix_time_from_system() * 1000),
				"elapsed_ms": elapsed_ms,
				"pipelines": _pipeline_counts(),
				"snapshot_equal": snapshot_equal,
				"nodes_before": nodes_before,
				"nodes_after": get_node_count(),
				"completed": prepared,
				"helper_only": true
			}
		)
	)
	paused = paused_before
