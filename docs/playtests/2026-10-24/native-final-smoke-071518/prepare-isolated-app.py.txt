def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def file_record(path):
    return {'path': str(path), 'bytes': path.stat().st_size, 'sha256': digest(path)}


def normalized_unsigned_engine(path):
    # codesign changes __LINKEDIT reservation sizes even after removing signatures.
    # Normalize only that load-command field; every other byte must still match.
    content = bytearray(path.read_bytes())
    assert content[:4] == bytes.fromhex('cafebabe'), 'Expected universal Mach-O'
    count = struct.unpack_from('>I', content, 4)[0]
    fields = []
    for index in range(count):
        _, _, offset, _, _ = struct.unpack_from('>IIIII', content, 8 + index * 20)
        assert content[offset:offset + 4] == bytes.fromhex('cffaedfe'), 'Expected little-endian Mach-O64'
        commands = struct.unpack_from('<I', content, offset + 16)[0]
        cursor = offset + 32
        found = False
        for _ in range(commands):
            command, size = struct.unpack_from('<II', content, cursor)
            assert size >= 8
            if command == 0x19 and content[cursor + 8:cursor + 24].rstrip(b'\0') == b'__LINKEDIT':
                position = cursor + 32
                fields.append({'offset': position, 'original_value': struct.unpack_from('<Q', content, position)[0]})
                struct.pack_into('<Q', content, position, 0)
                found = True
            cursor += size
        assert found, 'Missing __LINKEDIT load command'
    return {'sha256': hashlib.sha256(content).hexdigest(), 'normalized_linkedit_vmsize_fields': fields}

func _ready():

    var expected = EXPECTED_DIRECTORY

# Adding the isolated-data override is confined to this disposable app copy.

(archive / 'copy-codesign-verify.log.txt').write_text(verify.stdout)
assert verify.returncode == 0, 'Isolated app copy signature is invalid'

unsigned = []
for label, executable in [('original', original_files['executable']), ('isolated', app / 'Contents/MacOS/Open RTS')]:
    copy = stage / (label + '-unsigned-engine')
    shutil.copy2(executable, copy)
    result = subprocess.run(['codesign', '--remove-signature', str(copy)], text=True, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
    assert result.returncode == 0, 'Cannot compare unsigned engine code'
    unsigned.append({**file_record(copy), 'normalized': normalized_unsigned_engine(copy)})
assert unsigned[0]['normalized']['sha256'] == unsigned[1]['normalized']['sha256'], 'Isolation changed executable bytes beyond signatures and __LINKEDIT reservation metadata'
